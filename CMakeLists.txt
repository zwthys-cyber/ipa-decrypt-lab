cmake_minimum_required(VERSION 3.20)
project(ipa_decrypt_lab VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(ipa-engine src/main.c)
target_compile_options(ipa-engine PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
if(APPLE)
  target_compile_definitions(ipa-engine PRIVATE _DARWIN_C_SOURCE)
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME parser-contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/contracts.py $<TARGET_FILE:ipa-engine>)
